   702→/** Format a number as Brazilian currency (R$) */
   703→export function formatCurrency(value: number): string {
   704→  return value.toLocaleString("pt-BR", {
   705→    style: "currency",
   706→    currency: "BRL",
   707→    minimumFractionDigits: 0,
   708→    maximumFractionDigits: 0,
   709→  });
   710→}

   712→/** Format a number as percentage with 1 decimal place */
   713→export function formatPercentage(value: number): string {
   714→  return `${value.toFixed(1)}%`;
   715→}

   717→/** Format Gini index with 3 decimal places */
   718→export function formatGini(value: number): string {
   719→  return value.toFixed(3);

   722→/** Get latest value from a YearValuePair series */
   723→export function getLatestValue(data: YearValuePair[]): number {
   724→  return data[data.length - 1]?.value ?? 0;
   725→}
   726→
   727→/** Get value for a specific year from a YearValuePair series */
   728→export function getValueForYear(data: YearValuePair[], year: number): number | null {
   729→  const entry = data.find((d) => d.year === year);
   730→  return entry ? entry.value : null;
   731→}
   732→
   733→/** Calculate the change between the first and last values in a series */
   734→export function getTotalChange(data: YearValuePair[]): number {
   735→  if (data.length < 2) return 0;
   736→  return data[data.length - 1].value - data[0].value;
   737→}
   738→
   739→/** Calculate the percentage point change between first and last values */
   740→export function getPercentagePointChange(data: YearValuePair[]): number {
   741→  return getTotalChange(data);
   742→}
   743→
   744→/** Get the year with the maximum value in a series */
   745→export function getYearWithMaxValue(data: YearValuePair[]): { year: number; value: number } {
   746→  const maxEntry = data.reduce((prev, curr) => (curr.value > prev.value ? curr : prev), data[0]);
   747→  return maxEntry;
   748→}
   749→
   750→/** Get the year with the minimum value in a series */
   751→export function getYearWithMinValue(data: YearValuePair[]): { year: number; value: number } {
   752→  const minEntry = data.reduce((prev, curr) => (curr.value < prev.value ? curr : prev), data[0]);
   753→  return minEntry;
   754→}
